% ── Slide 5: Spelling / Keyword Normalization ───────────────────
\begin{frame}{Improvement: Spelling \& Keyword Normalization}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \small
    \textbf{Keyword Search Normalization} (\texttt{keyword\_search\_utils.py})
    \begin{itemize}
      \item \textbf{Unicode NFKC Normalization}: Standardizes character representations and lowercases all inputs.
      \item \textbf{Punctuation to Spaces}: Replaces hyphens (\texttt{-}) and underscores (\texttt{\_}) with spaces to unify compound words.
      \item \textbf{Compound Collapsing}: Generates collapsed and expanded variants (e.g., \texttt{open ai} $\leftrightarrow$ \texttt{openai}) for high recall.
      \item \textbf{Filename Stem Extraction}: Extracts clean, searchable stems from paths and URLs.
      \item \textbf{Index Enrichment}: Combines raw content, title, and filename variants to build a rich sparse index.
    \end{itemize}

    \diagrambox[scale=0.7, every node/.style={font=\tiny}]{

% ── Slide 6: Google Auth JWT Flow ───────────────────────────────
\begin{frame}{Feature: Google OAuth \& JWT Authentication}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \small
    \textbf{Authentication \& Session Flow}
    \begin{itemize}
      \item \textbf{Frontend Login}: User signs in via Google OAuth in the React UI and receives a secure \texttt{id\_token}.
      \item \textbf{Backend Verification}: The token is sent to \texttt{POST /v2/auth/google}. FastAPI verifies the token signature and audience with Google's API.
      \item \textbf{User Provisioning}: \texttt{AuthService} checks Neon Postgres. If the email is new, a user record is created on the fly.
      \item \textbf{JWT Generation}: A signed HS256 JWT is generated containing user claims (\texttt{sub}, \texttt{email}, \texttt{iat}, \texttt{exp}).
      \item \textbf{Access Control}: The client attaches the JWT as a Bearer token in the \texttt{Authorization} header for all protected V3 endpoints.
    \end{itemize}
